# Standalone module build: intentionally does not route any model graph yet.
cmake_minimum_required(VERSION 3.21)
project(llama_lsa_native_module C CXX)
option(LLAMA_LSA_NATIVE "Build experimental LSA kernels and map loader" OFF)
option(LLAMA_LSA_NATIVE_HIP "Build experimental native HIP kernels (no execution)" OFF)
if(NOT LLAMA_LSA_NATIVE)
    message(STATUS "Native LSA is disabled (default)")
    return()
endif()
set(LSA_TREE "${CMAKE_CURRENT_LIST_DIR}/../..")
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_HIP OFF CACHE BOOL "" FORCE)
add_subdirectory("${LSA_TREE}/ggml" ggml)
add_library(llama-lsa-native STATIC
    "${LSA_TREE}/src/llama-lsa-native.cpp" "${LSA_TREE}/src/llama-lsa-map.cpp")
target_compile_features(llama-lsa-native PUBLIC cxx_std_17)
target_include_directories(llama-lsa-native PUBLIC "${LSA_TREE}/src")
target_link_libraries(llama-lsa-native PUBLIC ggml-cpu ggml-base)
if(LLAMA_LSA_NATIVE_HIP)
    enable_language(HIP)
    find_package(hip REQUIRED)
    set_source_files_properties("${LSA_TREE}/src/llama-lsa-native-hip.hip" PROPERTIES LANGUAGE HIP)
    target_sources(llama-lsa-native PRIVATE "${LSA_TREE}/src/llama-lsa-native-hip.hip")
    target_compile_definitions(llama-lsa-native PRIVATE GGML_USE_HIP)
    target_include_directories(llama-lsa-native PRIVATE "${LSA_TREE}/ggml/src" "${LSA_TREE}/ggml/src/ggml-cuda")
    target_link_libraries(llama-lsa-native PUBLIC hip::host)
endif()
add_executable(test-lsa-native "${LSA_TREE}/tests/test-lsa-native.cpp")
target_link_libraries(test-lsa-native PRIVATE llama-lsa-native)
add_executable(test-lsa-map "${LSA_TREE}/tests/test-lsa-map.cpp")
target_link_libraries(test-lsa-map PRIVATE llama-lsa-native)
enable_testing()
add_test(NAME lsa-native COMMAND test-lsa-native)
add_test(NAME lsa-map COMMAND test-lsa-map "${CMAKE_CURRENT_BINARY_DIR}/map-fixture.bin")
add_executable(lsa-inspect-map inspect_map.cpp)
target_link_libraries(lsa-inspect-map PRIVATE llama-lsa-native)
if(LLAMA_LSA_NATIVE_HIP)
    add_executable(test-lsa-native-hip "${LSA_TREE}/tests/test-lsa-native-hip.cpp")
    target_link_libraries(test-lsa-native-hip PRIVATE llama-lsa-native)
    target_include_directories(test-lsa-native-hip PRIVATE "${LSA_TREE}/ggml/src")
    add_test(NAME lsa-native-hip-default-skip COMMAND test-lsa-native-hip)
    set_tests_properties(lsa-native-hip-default-skip PROPERTIES SKIP_RETURN_CODE 77)
    # Device execution still requires an explicit --run-hip and separate claims.
endif()
if(LLAMA_LSA_NATIVE_HIP)
    add_executable(lsa-inspect-hip-memory inspect_hip_memory.cpp)
    target_link_libraries(lsa-inspect-hip-memory PRIVATE llama-lsa-native)
    add_test(NAME lsa-hip-memory-host COMMAND lsa-inspect-hip-memory)
endif()
# Runtime lifecycle adapter is above the native kernels: no ggml/core callers.
add_library(llama-lsa-runtime STATIC "${LSA_TREE}/src/llama-lsa-runtime.cpp")
target_compile_features(llama-lsa-runtime PUBLIC cxx_std_17)
target_link_libraries(llama-lsa-runtime PUBLIC llama-lsa-native)
if(LLAMA_LSA_NATIVE_HIP)
    target_sources(llama-lsa-runtime PRIVATE "${LSA_TREE}/src/llama-lsa-runtime-hip.cpp")
    target_compile_definitions(llama-lsa-runtime PRIVATE LLAMA_LSA_RUNTIME_HIP)
endif()
add_executable(test-lsa-runtime "${LSA_TREE}/tests/test-lsa-runtime.cpp")
target_link_libraries(test-lsa-runtime PRIVATE llama-lsa-runtime)
add_test(NAME lsa-runtime COMMAND test-lsa-runtime)
if(LLAMA_LSA_NATIVE_HIP)
    add_executable(test-lsa-runtime-hip "${LSA_TREE}/tests/test-lsa-runtime-hip.cpp")
    target_link_libraries(test-lsa-runtime-hip PRIVATE llama-lsa-runtime)
    add_test(NAME lsa-runtime-hip-default-skip COMMAND test-lsa-runtime-hip)
    set_tests_properties(lsa-runtime-hip-default-skip PROPERTIES SKIP_RETURN_CODE 77)
endif()
